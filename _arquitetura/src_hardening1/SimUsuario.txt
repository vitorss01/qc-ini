Option Explicit
' ===== HARNESS DO AGENTE USUARIO =====
'
' Cada funcao simula UMA acao do analista e devolve "OK: ..." ou "FALHA: ...".
' Roda dentro do Excel, com os formularios de verdade carregados -- nao le
' codigo, exercita comportamento.
'
' Load frmX sem Show: o Initialize roda, os controles existem e ficam
' inspecionaveis, e nada aparece na tela. Atribuir valor a um controle DISPARA
' o evento Change, entao a reacao do formulario e observavel de fora.
'
' Modulo TEMPORARIO: inserido pela simulacao e removido ao final.

Private Function Falha(ByVal msg As String) As String
    Falha = "FALHA: " & msg
End Function

' ============================ frmCorrida ============================

Public Function S_Corrida_Carga() As String
    On Error GoTo erro
    Load frmCorrida
    S_Corrida_Carga = "OK: Initialize concluiu, " & frmCorrida.Controls.Count & " controles"
    Exit Function
erro:
    S_Corrida_Carga = Falha("erro " & Err.Number & " no Initialize: " & Err.Description)
End Function

Public Function S_Corrida_Niveis() As String
    On Error GoTo erro
    Load frmCorrida
    If frmCorrida.cboNivel.ListCount <> NLV Then
        S_Corrida_Niveis = Falha("combo tem " & frmCorrida.cboNivel.ListCount & _
                                 " niveis, produto tem " & NLV)
        Exit Function
    End If
    S_Corrida_Niveis = "OK: " & NLV & " niveis, coerente com o produto"
    Exit Function
erro:
    S_Corrida_Niveis = Falha("erro " & Err.Number & ": " & Err.Description)
End Function

' O analista trocou de lote na Configuracao e abre o formulario. Se o combo
' nao vier no lote ativo, ele lanca resultado no lote errado sem perceber --
' e o erro so aparece semanas depois, na estatistica.
Public Function S_Corrida_LoteAtivo() As String
    Dim ativo As String, sel As String
    On Error GoTo erro
    Load frmCorrida
    ativo = LoteAtivoCore()
    If frmCorrida.cboLote.ListCount = 0 Then
        S_Corrida_LoteAtivo = Falha("combo de lote vazio")
        Exit Function
    End If
    sel = CStr(frmCorrida.cboLote.Value)
    If sel <> ativo Then
        S_Corrida_LoteAtivo = Falha("lote ativo e '" & ativo & "' mas o combo abriu em '" & sel & "'")
        Exit Function
    End If
    S_Corrida_LoteAtivo = "OK: combo abriu no lote ativo '" & sel & "'"
    Exit Function
erro:
    S_Corrida_LoteAtivo = Falha("erro " & Err.Number & ": " & Err.Description)
End Function

Public Function S_Corrida_CamposAnalito() As String
    Dim n As Long, c As Collection
    On Error GoTo erro
    Load frmCorrida
    Set c = ListaAnalitos()
    On Error Resume Next
    n = 0
    Dim i As Long
    For i = 1 To c.Count
        If Not frmCorrida.Controls("txtV" & i) Is Nothing Then n = n + 1
    Next i
    On Error GoTo erro
    If n <> c.Count Then
        S_Corrida_CamposAnalito = Falha("cadastrados " & c.Count & " analitos, criados " & n & " campos")
        Exit Function
    End If
    S_Corrida_CamposAnalito = "OK: " & n & " campos, um por analito"
    Exit Function
erro:
    S_Corrida_CamposAnalito = Falha("erro " & Err.Number & ": " & Err.Description)
End Function

Public Function S_Excluir_Todos() As String
    Dim i As Long, n As Long
    On Error GoTo erro
    Load frmExcluir
    For i = 0 To frmExcluir.lstAnalitos.ListCount - 1
        frmExcluir.lstAnalitos.Selected(i) = True
    Next i
    For i = 0 To frmExcluir.lstAnalitos.ListCount - 1
        If frmExcluir.lstAnalitos.Selected(i) Then n = n + 1
    Next i
    If n <> frmExcluir.lstAnalitos.ListCount Then
        S_Excluir_Todos = Falha("selecionou " & n & " de " & frmExcluir.lstAnalitos.ListCount)
        Exit Function
    End If
    For i = 0 To frmExcluir.lstAnalitos.ListCount - 1
        frmExcluir.lstAnalitos.Selected(i) = False
    Next i
    S_Excluir_Todos = "OK: selecao multipla funciona (" & n & " itens)"
    Exit Function
erro:
    S_Excluir_Todos = Falha("erro " & Err.Number & ": " & Err.Description)
End Function

' Colagem com nivel 3 num produto de 2 niveis TEM de ser recusada. Aceitar
' criaria um nivel que nao existe, e o motor nao teria alvo nem DP para ele.
Public Function S_Massa_Validacao() As String
    Dim niveisAceitos As String, i As Long
    On Error GoTo erro
    niveisAceitos = ""
    For i = 1 To 4
        If i >= 1 And i <= NLV Then niveisAceitos = niveisAceitos & i
    Next i
    If NLV >= 3 Then
        S_Massa_Validacao = "OK: produto de " & NLV & " niveis, nada a recusar em 1..3"
        Exit Function
    End If
    ' confere que o codigo do formulario limita ao NLV do produto
    Dim cm As Object, txt As String
    Set cm = ThisWorkbook.VBProject.VBComponents("frmMassa").CodeModule
    txt = cm.Lines(1, cm.CountOfLines)
    If InStr(txt, "lvl > " & NLV) = 0 Then
        S_Massa_Validacao = Falha("validacao de nivel nao limita a " & NLV & _
                                  " - colagem com nivel inexistente seria aceita")
        Exit Function
    End If
    S_Massa_Validacao = "OK: colagem limitada a 1.." & NLV
    Exit Function
erro:
    S_Massa_Validacao = Falha("erro " & Err.Number & ": " & Err.Description)
End Function

' O nucleo do formulario: informar o RUN e o sistema preencher o resto.
Public Function S_NC_RunPreenche() As String
    Dim runs As Collection, r As Long
    On Error GoTo erro
    Load frmResultadoNaoConforme
    Set runs = RunsDoLote(LoteAtivoCore())
    If runs.Count = 0 Then
        S_NC_RunPreenche = Falha("nao ha RUN no lote ativo para simular")
        Exit Function
    End If
    r = runs(1)
    frmResultadoNaoConforme.txtRun.Value = CStr(r)
    If InStr(frmResultadoNaoConforme.lblCabecalho.Caption, "Data:") = 0 Then
        S_NC_RunPreenche = Falha("cabecalho nao preencheu: '" & _
                                 frmResultadoNaoConforme.lblCabecalho.Caption & "'")
        Exit Function
    End If
    If frmResultadoNaoConforme.lstAnalitos.ListCount = 0 Then
        S_NC_RunPreenche = Falha("RUN " & r & " nao trouxe resultado nenhum")
        Exit Function
    End If
    S_NC_RunPreenche = "OK: RUN " & r & " -> " & frmResultadoNaoConforme.lblCabecalho.Caption & _
                       " | " & frmResultadoNaoConforme.lstAnalitos.ListCount & " resultados"
    Exit Function
erro:
    S_NC_RunPreenche = Falha("erro " & Err.Number & ": " & Err.Description)
End Function

    Exit Function

Public Function S_NC_Todos() As String
    Dim runs As Collection, i As Long, n As Long
    On Error GoTo erro
    Load frmResultadoNaoConforme
    Set runs = RunsDoLote(LoteAtivoCore())
    If runs.Count = 0 Then S_NC_Todos = Falha("sem RUN para simular"): Exit Function
    frmResultadoNaoConforme.txtRun.Value = CStr(runs(1))
    For i = 0 To frmResultadoNaoConforme.lstAnalitos.ListCount - 1
        frmResultadoNaoConforme.lstAnalitos.Selected(i) = True
    Next i
    For i = 0 To frmResultadoNaoConforme.lstAnalitos.ListCount - 1
        If frmResultadoNaoConforme.lstAnalitos.Selected(i) Then n = n + 1
    Next i
    If n = 0 Or n <> frmResultadoNaoConforme.lstAnalitos.ListCount Then
        S_NC_Todos = Falha("selecao multipla nao funcionou (" & n & " de " & _
                           frmResultadoNaoConforme.lstAnalitos.ListCount & ")")
        Exit Function
    End If
    S_NC_Todos = "OK: " & n & " itens selecionados"
    Exit Function
erro:
    S_NC_Todos = Falha("erro " & Err.Number & ": " & Err.Description)
End Function
